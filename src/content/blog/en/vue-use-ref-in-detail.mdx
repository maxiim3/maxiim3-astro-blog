```js
function useRef(initialValue) {
  const _value = {
    value: initialValue ?? undefined,
  };

  return new Proxy(_value, {
    get(target, prop) {
      if (prop === "value") {
        return target[prop];
      } else console.error("prop doesnot exist");
    },
    set(target, prop, value) {
      if (prop === "value") {
        target[prop] = value;
        return true;
      }
      console.error("prop doesnot exist");
      return false;
    },
  });
}
```
